// FIX: PLAID – canonical implementation
import express from 'express';
import { Configuration, PlaidApi, PlaidEnvironments } from 'plaid';
export const plaidRouter = express.Router();
const env = (process.env.PLAID_ENV || 'sandbox') as keyof typeof PlaidEnvironments;
const cfg = new Configuration({
  basePath: PlaidEnvironments[env],
  baseOptions: { headers: {
    'PLAID-CLIENT-ID': process.env.PLAID_CLIENT_ID!,
    'PLAID-SECRET': process.env.PLAID_SECRET!
  }}
});
const plaid = new PlaidApi(cfg);

plaidRouter.get('/create_link_token', async (req, res) => {
  try {
    const userId = (req as any).session?.userId || 'demo-user-123';
    const r = await plaid.linkTokenCreate({
      user: { client_user_id: String(userId) },
      client_name: 'PocketTrack',
      language: 'en',
      country_codes: ['US'],
      products: ['transactions'],
      webhook: `${process.env.APP_BASE_URL}/api/plaid/webhook`,
      redirect_uri: `${process.env.APP_BASE_URL}/plaid/return`
    });
    res.json({ link_token: r.data.link_token });
  } catch (e:any) {
    console.error('PLAID create_link_token error', e?.response?.data || e);
    res.status(500).json({ error: 'link_token_failed' });
  }
});

plaidRouter.post('/exchange_public_token', async (req, res) => {
  try {
    const { public_token } = req.body || {};
    if (!public_token) return res.status(400).json({ error: 'missing_public_token' });
    const r = await plaid.itemPublicTokenExchange({ public_token });
    // TODO: persist r.data.access_token to your DB encrypted and associate to current user
    console.log('PLAID access_token obtained for user', (req as any).session?.userId || 'demo');
    res.json({ status: 'ok' });
  } catch (e:any) {
    console.error('PLAID exchange error', e?.response?.data || e);
    res.status(500).json({ error: 'exchange_failed' });
  }
});

plaidRouter.post('/webhook', (req, res) => {
  console.log('PLAID WEBHOOK:', req.body?.webhook_type, req.body?.webhook_code);
  res.sendStatus(200);
});

// FIX: PLAID – React button
import React from 'react';
import { usePlaidLink } from '@plaid/react-plaid-link';

export default function ConnectBankButton() {
  const [token, setToken] = React.useState<string>();
  React.useEffect(() => {
    fetch('/api/plaid/create_link_token', { credentials:'include' })
      .then(r => r.json()).then(d => setToken(d.link_token))
      .catch(() => setToken(undefined));
  }, []);
  const onSuccess = async (public_token: string) => {
    await fetch('/api/plaid/exchange_public_token', {
      method:'POST',
      headers:{ 'Content-Type':'application/json' },
      credentials:'include',
      body: JSON.stringify({ public_token })
    });
    window.location.href = '/';
  };
  const { open, ready } = usePlaidLink({ token, onSuccess });
  return <button data-action="connect-bank" disabled={!ready} onClick={()=>open()} className="btn btn-primary">Connect bank</button>;
}
