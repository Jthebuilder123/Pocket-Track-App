import express from 'express';
import { Configuration, PlaidApi, PlaidEnvironments } from 'plaid';
export const plaidRouter = express.Router();
const cfg = new Configuration({ basePath: PlaidEnvironments[process.env.PLAID_ENV as keyof typeof PlaidEnvironments], baseOptions: { headers: { 'PLAID-CLIENT-ID': process.env.PLAID_CLIENT_ID!, 'PLAID-SECRET': process.env.PLAID_SECRET! } }});
const plaid = new PlaidApi(cfg);

plaidRouter.get('/create_link_token', async (req, res) => {
  const userId = (req as any).user?.id ?? 'demo-user-123';
  const r = await plaid.linkTokenCreate({
    user: { client_user_id: String(userId) },
    client_name: 'PocketTrack',
    language: 'en',
    country_codes: ['US'],
    products: ['transactions'],
    webhook: `${process.env.APP_BASE_URL}/api/plaid/webhook`,
    redirect_uri: `${process.env.APP_BASE_URL}/plaid/return`
  });
  res.json({ link_token: r.data.link_token });
});

Create a hook component that fetches a link token and opens Link:

// src/client/components/ConnectBankButton.tsx
import React from 'react';
import { usePlaidLink } from '@plaid/react-plaid-link';

export default function ConnectBankButton() {
  const [token, setToken] = React.useState<string>();
  React.useEffect(() => { fetch('/api/plaid/create_link_token').then(r=>r.json()).then(d=>setToken(d.link_token)); }, []);
  const onSuccess = async (public_token: string) => {
    await fetch('/api/plaid/exchange_public_token', { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({ public_token }) });
    window.location.href = '/';
  };
  const { open, ready } = usePlaidLink({ token, onSuccess });
  return <button data-action="connect-bank" className="btn" onClick={()=>open()} disabled={!ready}>Connect bank</button>;
}

const PRICE_MAP: Record<string,string> = {
  'essentials:monthly': process.env.STRIPE_PRICE_E_MONTHLY!,
  'essentials:yearly':  process.env.STRIPE_PRICE_E_YEARLY!,
  'pro:monthly':        process.env.STRIPE_PRICE_P_MONTHLY!,
  'pro:yearly':         process.env.STRIPE_PRICE_P_YEARLY!,
};

billing.get('/checkout', async (req, res) => {
  const { plan='pro', interval='monthly' } = req.query as any;
  const price = PRICE_MAP[`${plan}:${interval}`];

  const user = (req as any).user || { id: 'demo-1', email: 'demo@example.com' };
  const session = await stripe.checkout.sessions.create({
    mode: 'subscription',
    customer_email: user.email,
    line_items: [{ price, quantity: 1 }],
    success_url: `${process.env.APP_BASE_URL}/billing/success?sid={CHECKOUT_SESSION_ID}`,
    cancel_url: `${process.env.APP_BASE_URL}/pricing`
  });
  res.redirect(303, session.url!);
});
